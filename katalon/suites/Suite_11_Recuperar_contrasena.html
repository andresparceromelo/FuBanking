CP06_Cambio_Contrasena_Exitoso
open http://localhost:3000/forgot-password
runScript localStorage.clear(); sessionStorage.clear(); document.cookie = "token=; path=/; expires=Thu, 01 Jan 1970 00:00:00 GMT";
pause 500
runScript fetch('http://localhost:3001/api/v1/auth/forgot-password', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ email: 'pruebaskatalon2@gmail.com' }) }).then(function(r) { return r.json(); }).then(function(res) { if (res && res.data && res.data.token) { window.location.href = 'http://localhost:3000/reset-password?token=' + res.data.token; } });
pause 4000
waitForElementPresent id=newPassword 5000
runScript const input = document.getElementById('newPassword'); const nativeSetter = Object.getOwnPropertyDescriptor(window.HTMLInputElement.prototype, 'value').set; nativeSetter.call(input, 'Katalon123.'); input.dispatchEvent(new Event('input', { bubbles: true })); input.dispatchEvent(new Event('change', { bubbles: true }));
pause 500
runScript const input = document.getElementById('confirmPassword'); const nativeSetter = Object.getOwnPropertyDescriptor(window.HTMLInputElement.prototype, 'value').set; nativeSetter.call(input, 'Katalon123.'); input.dispatchEvent(new Event('input', { bubbles: true })); input.dispatchEvent(new Event('change', { bubbles: true }));
pause 500
click css=button[type="submit"]
pause 3000
waitForElementPresent xpath=//*[contains(text(),'¡Contraseña actualizada!')] 5000
assertElementPresent xpath=//*[contains(text(),'¡Contraseña actualizada!')]
assertElementPresent xpath=//*[contains(text(),'Tu contraseña se ha cambiado correctamente')]
assertElementPresent xpath=//button[contains(text(),'Ir a Iniciar Sesión')]
click xpath=//button[contains(text(),'Ir a Iniciar Sesión')]
pause 1500
waitForElementPresent id=email 5000
assertLocation http://localhost:3000/login
